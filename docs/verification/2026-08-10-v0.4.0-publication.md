```python
import hashlib
import json
from pathlib import Path
import sys
from zipfile import ZipFile

archive = Path(sys.argv[1])
manifest = Path(sys.argv[2])
data = json.loads(manifest.read_text(encoding="utf-8"))
assert list(data) == ["archive", "archive_sha256", "files", "name", "version"]
assert data["name"] == "clinical-data-research-navigator"
assert data["version"] == "0.4.0"
assert data["archive"] == archive.name
archive_sha256 = hashlib.sha256(archive.read_bytes()).hexdigest()
manifest_sha256 = hashlib.sha256(manifest.read_bytes()).hexdigest()
assert archive_sha256 == data["archive_sha256"]
records = data["files"]
assert len(records) == 8
assert all(list(record) == ["path", "sha256", "size"] for record in records)
paths = [record["path"] for record in records]
assert paths == sorted(paths)
assert len(paths) == len(set(paths))
with ZipFile(archive) as zip_file:
    infos = zip_file.infolist()
    names = [info.filename for info in infos]
    assert len(names) == len(set(names))
    assert names == paths
    members = []
    for info, record in zip(infos, records, strict=True):
        payload = zip_file.read(info)
        member_sha256 = hashlib.sha256(payload).hexdigest()
        assert info.file_size == len(payload) == record["size"]
        assert member_sha256 == record["sha256"]
        members.append(
            {"path": info.filename, "size": len(payload), "sha256": member_sha256}
        )
print(
    json.dumps(
        {
            "root_keys": list(data),
            "version": data["version"],
            "archive": data["archive"],
            "archive_sha256": archive_sha256,
            "manifest_sha256": manifest_sha256,
            "member_count": len(members),
            "members": members,
        },
        ensure_ascii=False,
        indent=2,
    )
)
```
